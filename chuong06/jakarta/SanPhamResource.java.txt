@Path("/sanpham")
public class SanPhamResource {

    @Inject                       // CDI tiêm phụ thuộc thay cho lời gọi new
    private SanPhamService dichVu;

    @GET
    @Produces(MediaType.APPLICATION_JSON)
    public List<SanPham> danhSach() throws SQLException {
        return dichVu.laySapXepTheoGiaGiamDan();
    }
}
